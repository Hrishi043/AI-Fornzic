# Notebook code cell 9
chunk_size = 25_000

def process_activity_file(filename):
    file_path = data_dir / filename
    source = file_path.stem.lower()

    # Read the header first so we can select only useful columns.
    available_columns = pd.read_csv(file_path, nrows=0).columns.tolist()
    available_columns = [column.strip().lower() for column in available_columns]

    useful_columns = ["date", "user", "pc", "activity", "attachments", "filename"]
    columns_to_read = [
        column for column in useful_columns
        if column in available_columns
    ]

    if "date" not in columns_to_read or "user" not in columns_to_read:
        raise ValueError(f"{filename} needs both 'date' and 'user' columns.")

    daily_parts = []
    quality = {
        "rows_read": 0,
        "rows_kept": 0,
        "missing_user": 0,
        "invalid_date": 0,
        "duplicates_removed": 0,
    }

    for chunk in pd.read_csv(
        file_path,
        usecols=lambda column: column.strip().lower() in columns_to_read,
        chunksize=chunk_size,
        low_memory=False,
    ):
        chunk.columns = [column.strip().lower() for column in chunk.columns]
        quality["rows_read"] += len(chunk)

        chunk["user"] = chunk["user"].astype("string").str.strip()
        missing_user = chunk["user"].isna() | chunk["user"].eq("")
        quality["missing_user"] += int(missing_user.sum())

        dates = pd.to_datetime(chunk["date"], errors="coerce", format="mixed")
        invalid_date = dates.isna()
        quality["invalid_date"] += int(invalid_date.sum())

        duplicate = chunk.duplicated()
        quality["duplicates_removed"] += int(duplicate.sum())

        keep = ~(missing_user | invalid_date | duplicate)
        chunk = chunk.loc[keep].copy()
        dates = dates.loc[keep]

        if chunk.empty:
            continue

        quality["rows_kept"] += len(chunk)

        # A row is outside normal working hours if it is before 6 AM
        # or at/after 6 PM, based on the timestamp in the dataset.
        hour = dates.dt.hour
        chunk["after_hours_count"] = (
            (hour < 6) | (hour >= 18)
        ).astype("int8").to_numpy()

        chunk["day"] = dates.dt.strftime("%Y-%m-%d").to_numpy()
        chunk["event_count"] = 1

        feature_columns = ["event_count", "after_hours_count"]

        if source == "device" and "activity" in chunk.columns:
            chunk["device_connect_count"] = (
                chunk["activity"].astype("string").str.lower() == "connect"
            ).astype("int8")
            feature_columns.append("device_connect_count")

        if source == "email" and "attachments" in chunk.columns:
            chunk["attachment_count"] = (
                pd.to_numeric(chunk["attachments"], errors="coerce")
                .fillna(0)
            )
            feature_columns.append("attachment_count")

        group = chunk.groupby(["user", "day"])[feature_columns].sum()
        group = group.rename(
            columns={column: f"{source}_{column}" for column in feature_columns}
        )
        daily_parts.append(group)

    if daily_parts:
        daily_features = (
            pd.concat(daily_parts)
            .groupby(level=["user", "day"])
            .sum()
        )
    else:
        daily_features = pd.DataFrame()

    return daily_features, quality

# Notebook code cell 25
target_column = "known_incident_window"

# Do not let employee IDs, dates, scenarios, or the answer label become model inputs.
excluded_columns = {
    "user",
    "day",
    "scenario",
    target_column,
}

    column for column in employee_daily.select_dtypes(include="number").columns
    if column not in excluded_columns
]

print("Features:", feature_columns)

# Hold out employees, not individual days, to avoid the same employee appearing
# in both model development and evaluation data.
outer_splitter = StratifiedGroupKFold(
    n_splits=5,
    shuffle=True,
    random_state=42,
)

train_validation_indices, test_indices = next(
    outer_splitter.split(X, y, groups=groups)
)

inner_splitter = StratifiedGroupKFold(
    n_splits=4,
    shuffle=True,
    random_state=43,
)
train_relative_indices, validation_relative_indices = next(
    inner_splitter.split(
        X.iloc[train_validation_indices],
        y.iloc[train_validation_indices],
        groups=groups.iloc[train_validation_indices],
    )
)
train_indices = train_validation_indices[train_relative_indices]
validation_indices = train_validation_indices[validation_relative_indices]

X_train = X.iloc[train_indices]
X_validation = X.iloc[validation_indices]
X_test = X.iloc[test_indices]
y_train = y.iloc[train_indices]
y_validation = y.iloc[validation_indices]
y_test = y.iloc[test_indices]

train_users = set(groups.iloc[train_indices])
validation_users = set(groups.iloc[validation_indices])
test_users = set(groups.iloc[test_indices])

print("Training employees:", len(train_users))

# Notebook code cell 26
models = {
    "Logistic Regression": make_pipeline(
        SimpleImputer(strategy="median"),
        StandardScaler(),
        LogisticRegression(
            max_iter=1000,
            class_weight="balanced",
            random_state=42,
        ),
    ),

    "Decision Tree": DecisionTreeClassifier(
        max_depth=10,
        min_samples_leaf=10,
        class_weight="balanced",
        random_state=42,
    ),

    "Random Forest": RandomForestClassifier(
        n_estimators=200,
        min_samples_leaf=5,
        class_weight="balanced_subsample",
        n_jobs=-1,
        random_state=42,
    ),

    # This model learns what is unusual without using the incident labels.
    "Isolation Forest": IsolationForest(
        n_estimators=200,
        contamination="auto",
        random_state=42,
        n_jobs=-1,
    ),
}

validation_risk_scores = {}
risk_scores = {}

for name, model in models.items():
    print(f"Training {name}...")
    model.fit(X_train, y_train)

    if name == "Isolation Forest":
        # Larger scores should mean more unusual activity.
        validation_risk_scores[name] = -model.decision_function(X_validation)
        risk_scores[name] = -model.decision_function(X_test)
    else:
        validation_risk_scores[name] = model.predict_proba(X_validation)[:, 1]
        risk_scores[name] = model.predict_proba(X_test)[:, 1]

print("All four models are trained and scored on validation and test employees.")

# Notebook code cell 28
from sklearn.metrics import (
    accuracy_score,
    average_precision_score,
    confusion_matrix,
    precision_score,
    recall_score,
    f1_score,
    ConfusionMatrixDisplay,
)

# Choose a practical review capacity on validation data only. Selecting from a
# coarse grid is less sensitive to one exact cutoff than searching every row.
candidate_review_rates = [0.01, 0.02, 0.03, 0.05, 0.075, 0.10, 0.15, 0.20]
validation_labels = y_validation.to_numpy()
actual_labels = y_test.to_numpy()

# Preserve model selection by validation Average Precision.
selected_model_for_rate = max(
    validation_risk_scores,
    key=lambda name: average_precision_score(
        validation_labels, np.asarray(validation_risk_scores[name])
    ),
)
selected_model_validation_scores = np.asarray(
    validation_risk_scores[selected_model_for_rate]
)

best_rate = None
best_validation_f1 = -1.0
for candidate_rate in candidate_review_rates:
    candidate_count = max(
        1, int(np.ceil(len(validation_labels) * candidate_rate))
    )
    candidate_indices = np.argsort(selected_model_validation_scores)[-candidate_count:]
    candidate_predictions = np.zeros(len(validation_labels), dtype=int)
    candidate_predictions[candidate_indices] = 1
    candidate_f1 = f1_score(
        validation_labels, candidate_predictions, zero_division=0
    )
    # Rates are ascending, so retaining the first tie favors the smaller queue.
    if candidate_f1 > best_validation_f1:
        best_validation_f1 = candidate_f1
        best_rate = candidate_rate

selected_queue_rate = float(best_rate)
validation_top_count = max(
    1, int(np.ceil(len(validation_labels) * selected_queue_rate))
)
test_top_count = max(1, int(np.ceil(len(actual_labels) * selected_queue_rate)))
validation_five_percent_count = max(
    1, int(np.ceil(len(validation_labels) * 0.05))
)
test_five_percent_count = max(1, int(np.ceil(len(actual_labels) * 0.05)))

print(
    f"Selected model: {selected_model_for_rate}; "
    f"validation-F1 queue capacity: {selected_queue_rate:.1%} "
    f"({validation_top_count:,} of {len(validation_labels):,} employee-days); "
    f"validation F1: {best_validation_f1:.3f}"
)

results = []
confusion_matrices = {}
for name, test_scores in risk_scores.items():
    validation_scores = np.asarray(validation_risk_scores[name])
    test_scores = np.asarray(test_scores)

    validation_top_indices = np.argsort(validation_scores)[-validation_top_count:]
    validation_predictions = np.zeros(len(validation_labels), dtype=int)
    validation_predictions[validation_top_indices] = 1
    test_top_indices = np.argsort(test_scores)[-test_top_count:]
    test_predictions = np.zeros(len(actual_labels), dtype=int)
    test_predictions[test_top_indices] = 1

    # Keep the original 5% capacity as a reference for the precision/recall tradeoff.
    validation_five_indices = np.argsort(validation_scores)[-validation_five_percent_count:]
    validation_five_predictions = np.zeros(len(validation_labels), dtype=int)
    validation_five_predictions[validation_five_indices] = 1
    test_five_indices = np.argsort(test_scores)[-test_five_percent_count:]
    test_five_predictions = np.zeros(len(actual_labels), dtype=int)
    test_five_predictions[test_five_indices] = 1

    test_matrix = confusion_matrix(
        actual_labels, test_predictions, labels=[0, 1]
    )
    confusion_matrices[name] = test_matrix

    results.append({
        "Model": name,
        "Selected queue rate": selected_queue_rate,
        "Validation AP": average_precision_score(
            validation_labels, validation_scores
        ),
        "Validation precision @ selected queue": precision_score(
            validation_labels, validation_predictions, zero_division=0
        ),
        "Validation recall @ selected queue": recall_score(
            validation_labels, validation_predictions, zero_division=0
        ),
        "Validation F1 @ selected queue": f1_score(
            validation_labels, validation_predictions, zero_division=0
        ),
        "Test precision @ selected queue": precision_score(
            actual_labels, test_predictions, zero_division=0
        ),
        "Test recall @ selected queue": recall_score(
            actual_labels, test_predictions, zero_division=0
        ),
        "Test F1 @ selected queue": f1_score(
            actual_labels, test_predictions, zero_division=0
        ),
        "Test false alerts @ selected queue": int(test_matrix[0, 1]),
        "Test AP": average_precision_score(actual_labels, test_scores),
        "Validation precision @ 5%": precision_score(
            validation_labels, validation_five_predictions, zero_division=0
        ),
        "Validation recall @ 5%": recall_score(
            validation_labels, validation_five_predictions, zero_division=0
        ),
        "Test precision @ 5%": precision_score(
            actual_labels, test_five_predictions, zero_division=0
        ),
        "Test recall @ 5%": recall_score(
            actual_labels, test_five_predictions, zero_division=0
        ),
        "Test accuracy @ 5%": accuracy_score(actual_labels, test_five_predictions),
        "Test F1 @ 5%": f1_score(
            actual_labels, test_five_predictions, zero_division=0
        ),
    })

# Select the model using validation AP; never use test metrics for selection.
model_comparison = (
